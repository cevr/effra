package check

// zz_parity_hook.go is written by scripts/parity196/run.sh into a scratch
// worktree only. It records every source the checker reads to the end,
// with its diagnostics, as one JSON line in $ERGO_PARITY_LOG, once per
// distinct source, with the test function that checked it first.

import (
	"crypto/sha256"
	"encoding/json"
	"os"
	"runtime"
	"strings"
	"sync"
)

var parity struct {
	sync.Mutex
	seen map[[32]byte]bool
}

type parityDiag struct {
	Code    string `json:"code"`
	Line    int    `json:"line"`
	Column  int    `json:"column"`
	Message string `json:"message"`
}

type parityRec struct {
	Test  string       `json:"test"`
	Src   string       `json:"src"`
	Diags []parityDiag `json:"diags"`
}

func parityTest() string {
	pcs := make([]uintptr, 64)
	n := runtime.Callers(3, pcs)
	frames := runtime.CallersFrames(pcs[:n])
	for {
		f, more := frames.Next()
		name := f.Function[strings.LastIndex(f.Function, "/")+1:]
		if _, rest, ok := strings.Cut(name, "."); ok {
			if strings.HasPrefix(rest, "Test") || strings.HasPrefix(rest, "Fuzz") {
				if i := strings.IndexByte(rest, '.'); i >= 0 {
					rest = rest[:i]
				}
				return rest
			}
		}
		if !more {
			return "?"
		}
	}
}

func parityRecord(src string, diags []Diagnostic) {
	path := os.Getenv("ERGO_PARITY_LOG")
	if path == "" {
		return
	}
	key := sha256.Sum256([]byte(src))
	parity.Lock()
	defer parity.Unlock()
	if parity.seen == nil {
		parity.seen = map[[32]byte]bool{}
	}
	if parity.seen[key] {
		return
	}
	parity.seen[key] = true
	rec := parityRec{Test: parityTest(), Src: src}
	for _, d := range diags {
		rec.Diags = append(rec.Diags, parityDiag{d.Code, d.Line, d.Column, d.Message})
	}
	line, err := json.Marshal(rec)
	if err != nil {
		return
	}
	f, err := os.OpenFile(path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
	if err != nil {
		return
	}
	f.Write(append(line, '\n'))
	f.Close()
}

func (c *checker) sorted() []Diagnostic {
	d := c.sortedUnhooked()
	if len(c.files) == 1 && c.files[0].syntax != nil {
		parityRecord(c.files[0].syntax.Src, d)
	}
	return d
}

// CheckSource records a source the parser refuses; Check records the rest.
func CheckSource(src string) (*Package, []Diagnostic) {
	p, d := checkSourceUnhooked(src)
	if p == nil {
		parityRecord(src, d)
	}
	return p, d
}
